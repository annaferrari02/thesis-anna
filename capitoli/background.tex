\section{Hemodynamics in AAA and after EVAR}

\subsection{Wall shear stress indices}
\label{sec:wss-indices}
Let $\boldsymbol{\tau}_w(\mathbf{x},t)$ be the wall shear stress at wall point $\mathbf{x}$, and $[0,T]$ the last cardiac cycle.
All indices are averaged over the cycle \cite{mutlu_integrated_2026, katsoudas_hemodynamic_2026}.
\begin{equation}
    \mathrm{TAWSS}(\mathbf{x}) = \frac{1}{T}\int_0^T \lVert \boldsymbol{\tau}_w(\mathbf{x},t) \rVert \, dt
\end{equation}
\begin{equation}
    \mathrm{OSI}(\mathbf{x}) = \frac{1}{2}\left(1 - \frac{\left\lVert \int_0^T \boldsymbol{\tau}_w(\mathbf{x},t)\,dt \right\rVert}{\int_0^T \lVert \boldsymbol{\tau}_w(\mathbf{x},t) \rVert \, dt}\right)
\end{equation}
\begin{equation}
    \mathrm{RRT}(\mathbf{x}) = \frac{1}{\left(1 - 2\,\mathrm{OSI}(\mathbf{x})\right)\mathrm{TAWSS}(\mathbf{x})},
    \qquad
    \mathrm{ECAP}(\mathbf{x}) = \frac{\mathrm{OSI}(\mathbf{x})}{\mathrm{TAWSS}(\mathbf{x})}
\end{equation}
\begin{itemize}
    \item TAWSS: mean intensity of wall shear stress.
    \item OSI: change of direction; 0 = unidirectional, 0.5 = purely oscillatory. \citeneeded{Ku et al.\ 1985}
    \item RRT: high where shear stress is low and oscillatory (stagnation). \citeneeded{Himburg et al.\ 2004}
    \item ECAP: exposure of the wall to low and oscillatory shear, linked to thrombus formation. \citeneeded{Di Achille et al.\ 2014}
\end{itemize}

\subsection{The role of Hemodynamics in AAA studies}
Why hemodynamics could be informative for post-EVAR complications:
\begin{itemize}
    \item Links between WSS indices and thrombus in AAA, as well as AAA growth or rupture \cite{qiu_association_2022, zambrano_association_2022, belkacemi_intraluminal_2023}.
    \item Load on the proximal neck and landing zones $\rightarrow$ seal failure (type I endoleak, migration). \citeneeded{mechanism}
    \item Low and oscillatory flow in the sac $\rightarrow$ thrombus, branch backflow (type II endoleak, sac behaviour). \citeneeded{mechanism}
\end{itemize}

\subsection{Learning on point clouds}
\begin{itemize}
    \item PointNet: shared MLP on each point and max pooling; invariant to point order. \citeneeded{Qi et al.\ 2017, PointNet}
    \item PointNet++: the same idea applied on local neighbourhoods, from local to global features. \citeneeded{Qi et al.\ 2017, PointNet++}
    \item Point-based methods use Euclidean neighbourhoods, not the mesh connectivity; mesh-based alternatives exist. \citeneeded{Pfaff et al.\ 2021, MeshGraphNets}
    \item Spatial patterns of hemodynamic indices in AAA \cite{rezaeitaleshmahalleh_characterization_2023}.
\end{itemize}

\section{Closest work and gap}
\begin{itemize}
    \item Peng et al.: point-cloud network with morphology, radiomics and CFD; AUC from 0.81 to 0.93. \citeneeded{Peng et al.\ 2026, medRxiv}
    \item The +0.12 AUC from CFD is mixed with an architecture change (1 vs.\ 3 feature extractors). It cannot show that CFD itself improved the prediction.
    \item One test set, no confidence intervals or paired tests: no uncertainty quantification.
    \item Gap: no study isolates the contribution of preoperative CFD in risk prediction of post EVAR complication with a fixed model.
    \item Regional means may lose where on the wall the indices are high or low; a point-based encoder can keep this spatial information. \citeneeded{Qi et al.\ 2017, PointNet++}
\end{itemize}

\subsection{Testing a new predictor}
\begin{itemize}
    \item Added value of a new marker \cite{cook_quantifying_2018}.
    \item For nested models, ``no added information'' is the same null hypothesis for the coefficients, $\Delta$AUC, net benefit, NRI and IDI \cite{pepe_testing_2013}.
    \item Test the coefficients with a likelihood ratio test; report $\Delta$AUC as an effect size. \citeneeded{Vickers, Cronin, Begg 2011}

\begin{itemize}
    \item Recalibration and extension of an existing model with new predictors \cite{janssen_updating_2008}.

\subsection{Research questions and hypotheses}
\textbf{Q1.} Do preoperative hemodynamic features carry predictive information about post-EVAR adverse events beyond the reference radiomic--morphological model?

Formal hypothesis, following \cite{pepe_testing_2013}:
\begin{align}
    H_0:&\; P\left(D = 1 \mid S_B, \mathbf{Hd}\right) = P\left(D = 1 \mid S_B\right) \\
    H_1:&\; \mathbf{Hd} \text{ carries additional information about } D \text{ given } S_B
\end{align}
Q2 starts from a limit of $\mathbf{Hd}$: two patients with the same mean TAWSS in the sac have the same $\mathbf{Hd}$,
even if in one the low TAWSS is diffuse and in the other it is concentrated near the lumbar arteries.
Only a representation that keeps the position of the values can separate them.

\textbf{Q2.} Does a geometry-aware representation $\mathbf{Z}$ of the same indices, learned by a point-based encoder, carry information beyond $S_B$ and $\mathbf{Hd}$?
\begin{align}
    H_0^{(2)}:&\; P\left(D = 1 \mid S_B, \mathbf{Hd}, \mathbf{Z}\right) = P\left(D = 1 \mid S_B, \mathbf{Hd}\right) \\
    H_1^{(2)}:&\; \mathbf{Z} \text{ carries additional information about } D \text{ given } S_B \text{ and } \mathbf{Hd}
\end{align}
Both questions are purely predictive (conditional association), not causal.
