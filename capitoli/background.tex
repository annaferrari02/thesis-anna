\section{CFD and wall shear stress indices}

\subsection{Indices}
\label{sec:wss-indices}
Let $\boldsymbol{\tau}_w(\mathbf{x},t)$ be the wall shear stress at wall point $\mathbf{x}$, and $[0,T]$ the last cardiac cycle.
All indices are averaged over the cycle, so they do not depend on the cardiac phase
\cite{mutlu_integrated_2026, katsoudas_hemodynamic_2026}.

\begin{equation}
    \mathrm{TAWSS}(\mathbf{x}) = \frac{1}{T}\int_0^T \lVert \boldsymbol{\tau}_w(\mathbf{x},t) \rVert \, dt
\end{equation}

\begin{equation}
    \mathrm{OSI}(\mathbf{x}) = \frac{1}{2}\left(1 - \frac{\left\lVert \int_0^T \boldsymbol{\tau}_w(\mathbf{x},t)\,dt \right\rVert}{\int_0^T \lVert \boldsymbol{\tau}_w(\mathbf{x},t) \rVert \, dt}\right)
\end{equation}

\begin{equation}
    \mathrm{RRT}(\mathbf{x}) = \frac{1}{\left(1 - 2\,\mathrm{OSI}(\mathbf{x})\right)\mathrm{TAWSS}(\mathbf{x})}
\end{equation}

\begin{itemize}
    \item TAWSS: mean intensity of wall shear stress.
    \item OSI: change of direction; 0 = unidirectional, 0.5 = purely oscillatory. \citeneeded{Ku et al.\ 1985}
    \item RRT: high where shear stress is low and oscillatory (stagnation). \citeneeded{Himburg et al.\ 2004}
\end{itemize}

\section{Incremental value of a new marker}
\label{sec:framework}

\subsection{Null hypothesis}
\begin{itemize}
    \item Question: given an existing model, does a new marker change the risk estimate? \cite{cook_quantifying_2018}
\end{itemize}
\begin{equation}
    H_0:\; P\left(y = 1 \mid S_B, \mathbf{Hd}\right) = P\left(y = 1 \mid S_B\right)
\end{equation}
\begin{itemize}
    \item For nested models, $\boldsymbol{\gamma}=\mathbf{0}$ is equivalent to no gain in AUC, net benefit, NRI and IDI \cite{pepe_testing_2013}.
    \item One test is enough: the likelihood ratio test (LRT).
\end{itemize}

\subsection{Why not a test on $\Delta$AUC}
\begin{itemize}
    \item For nested models, the DeLong test on AUC is very conservative and has low power; LRT and Wald keep the nominal level. \citeneeded{Vickers, Cronin, Begg 2011}
    \item Reason: the AUC test assumes two distinct markers, not nested models. \citeneeded{Begg, Gönen, Seshan 2013}
    \item Critical view of these simulations. \citeneeded{On the assessment of the added value of new predictive biomarkers, 2013}
    \item $\Delta$AUC is reported as an effect size, not as a test.
\end{itemize}

\subsection{Recalibration and model updating}
\begin{itemize}
    \item Discrimination vs.\ calibration \cite{alba_discrimination_2017}. \citeneeded{Van Calster et al.\ 2019}
    \item Logistic recalibration of an existing score: intercept and slope. \citeneeded{Cox 1958}
    \item Recalibration, revision and extension with new predictors \cite{janssen_updating_2008}. \citeneeded{Steyerberg 2019}
    \item Classical updating uses a separate update set.
\end{itemize}

\subsection{Pre-validation}
\begin{itemize}
    \item Out-of-fold predictions act as an independent update set for each patient. \citeneeded{Tibshirani and Efron 2002}
    \item Same idea as stacking. \citeneeded{Wolpert 1992}
    \item Known limit: the one-degree-of-freedom test can be biased. \citeneeded{Höfling and Tibshirani 2008}
\end{itemize}
