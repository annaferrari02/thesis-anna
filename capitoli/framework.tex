\section{Incremental value of a new marker}
\begin{itemize}
    \item Question: given an existing model, does a new marker change the risk estimate? \cite{cook_quantifying_2018}
    \item Null hypothesis in probability form:
\end{itemize}
\begin{equation}
    H_0:\; P\left(y = 1 \mid S_B, \mathbf{Hd}\right) = P\left(y = 1 \mid S_B\right)
\end{equation}

\section{Equivalent null hypotheses}
\begin{itemize}
    \item For nested models, $\boldsymbol{\gamma}=\mathbf{0}$ is equivalent to no gain in AUC, net benefit, NRI and IDI \cite{pepe_testing_2013}.
    \item One test is enough: the likelihood ratio test (LRT).
\end{itemize}

\section{Why not a test on $\Delta$AUC}
\begin{itemize}
    \item For nested models, the DeLong test on AUC is very conservative and has low power; LRT and Wald keep the nominal level. \citeneeded{Vickers, Cronin, Begg 2011}
    \item Reason: the AUC test assumes two distinct markers, not nested models. \citeneeded{Begg, Gönen, Seshan 2013}
    \item Critical view of these simulations. \citeneeded{On the assessment of the added value of new predictive biomarkers, 2013}
    \item $\Delta$AUC is reported as an effect size, not as a test.
\end{itemize}
