\section{Protocol}
\begin{itemize}
    \item Fixed before the analysis: outcome, $\mathbf{Hd}$ set, number of components $k$, models, endpoints.
    \item Sample size: events per parameter, with $E$ the number of events. \citeneeded{Peduzzi et al.\ 1996} \citeneeded{Riley et al.\ 2020}
\end{itemize}
\begin{equation}
    1 + p + k \;\le\; \frac{E}{10}
\end{equation}

\section{Reference model and score}
\begin{itemize}
    \item Reproduce the benchmark pipeline (Appendix~\ref{app:benchmark}).
    \item Out-of-fold score: 5 stratified folds, $R$ seeds, whole pipeline refitted in each fold. \citeneeded{Tibshirani and Efron 2002}
    \item Why out-of-fold: in-sample scores are optimistic and would bias the test against $\mathbf{Hd}$.
\end{itemize}
\begin{equation}
    \bar{p}_i = \frac{1}{R}\sum_{r=1}^{R} \hat{p}_i^{(r)},
    \qquad
    S_{B,i} = \operatorname{logit}(\bar{p}_i)
\end{equation}

\section{Primary analysis}

\subsection{Models}
\begin{align}
    M_0:\;& \operatorname{logit} P(D_i = 1) = \beta_0 + \beta_1 S_{B,i} \\
    M_1:\;& \operatorname{logit} P(D_i = 1) = \beta_0 + \beta_1 S_{B,i} + \boldsymbol{\gamma}^{\top}\mathbf{Hd}_i
\end{align}
$M_0$ is the recalibrated benchmark \citeneeded{Cox 1958}; $M_1$ extends it with $\mathbf{Hd}$ \cite{janssen_updating_2008}.

\subsection{Order of the analysis}
\begin{enumerate}
    \item \textbf{Calibration} of both models: intercept and slope \cite{alba_discrimination_2017}.
    \item \textbf{Likelihood ratio test} on the $\mathbf{Hd}$ block \cite{pepe_testing_2013}:
    \begin{equation}
        H_0:\; \boldsymbol{\gamma} = \mathbf{0},
        \qquad
        \Lambda = 2\left(\ell_1 - \ell_0\right) \;\overset{H_0}{\sim}\; \chi^2_{p}
    \end{equation}
    \item \textbf{Performance gain} with bootstrap CI: $\Delta$AUC, $\Delta$Brier, $\Delta$net benefit. {[}Thresholds $p_t$: to discuss with clinicians.]
    \begin{equation}
        \mathrm{NB}(p_t) = \frac{\mathrm{TP}(p_t)}{n} - \frac{\mathrm{FP}(p_t)}{n}\cdot\frac{p_t}{1 - p_t}
    \end{equation}
    \item \textbf{Correction for optimism}: out-of-fold estimates or bootstrap optimism correction. \citeneeded{Steyerberg 2019}
\end{enumerate}

\section{Geometry-aware analysis (Q2)}

\subsection{Point-based encoder}
\begin{itemize}
    \item Encoder $E_\phi$: small PointNet++. \citeneeded{Qi et al.\ 2017, PointNet++}
    \item Decoder $D_\psi$: receives the latent code and a coordinate, returns the indices at that point.
    \item Trained without the outcome (no label leakage).
\end{itemize}
\begin{equation}
    \mathbf{z}_i = E_\phi(\mathcal{P}_i) \in \mathbb{R}^{32},
    \qquad
    \hat{\mathbf{q}}_{ij} = D_\psi(\mathbf{z}_i, \mathbf{x}_{ij})
\end{equation}
\begin{equation}
    \mathcal{L}(\phi,\psi) = \frac{1}{nN}\sum_{i=1}^{n}\sum_{j=1}^{N} \left\lVert \hat{\mathbf{q}}_{ij} - \mathbf{q}_{ij} \right\rVert^2_{\mathbf{W}}
\end{equation}
\begin{itemize}
    \item Checks: reconstruction $R^2$ on unseen patients; linear probe $\mathbf{Hd}_i \approx \mathbf{A}\mathbf{z}_i + \mathbf{b}$.
\end{itemize}

\subsection{Reduction}
\begin{equation}
    \mathbf{Z}_i = \mathbf{V}_k^{\top}\left(\bar{\mathbf{z}}_i - \boldsymbol{\mu}\right) \in \mathbb{R}^{k}
\end{equation}
$\bar{\mathbf{z}}_i$: mean of $\mathbf{z}_i$ over point resamplings; PCA keeps $k$ components, fixed in advance.

\subsection{Models and test}
\begin{equation}
    M_3:\; \operatorname{logit} P(D_i = 1) = \beta_0 + \beta_1 S_{B,i} + \boldsymbol{\gamma}^{\top}\mathbf{Hd}_i + \boldsymbol{\delta}^{\top}\mathbf{Z}_i
\end{equation}
\begin{equation}
    H_0^{(2)}:\; \boldsymbol{\delta} = \mathbf{0},
    \qquad
    \Lambda_2 = 2\left(\ell_3 - \ell_1\right) \;\overset{H_0}{\sim}\; \chi^2_{k}
\end{equation}
\begin{itemize}
    \item Same order as the primary analysis: calibration, LRT, performance gain with CI.
    \item Encoder and PCA fitted once on the full cohort for the LRT; refitted inside each fold for the performance estimates.
    \item Holm--Bonferroni correction across Q1 and Q2.
\end{itemize}

\subsection{Controls and interpretation}
\begin{itemize}
    \item $\mathbf{Z}_{\text{shape}}$: same encoder on geometric descriptors only; separates hemodynamics from shape.
    \item $\mathbf{Z}_{\text{perm}}$: fields assigned to another patient's geometry; expected no gain.
    \item Each component decoded on a reference geometry to show its spatial pattern.
\end{itemize}

\section{Secondary and sensitivity analyses}
\begin{itemize}
    \item ML benchmark retrained with and without $\mathbf{Hd}$: paired $\Delta$AUC (effect size, not a test).
    \item $\mathbf{Hd}$-only model.
    \item Conditional permutation of $\mathbf{Hd}$. \citeneeded{conditional permutation test}
    \item Alternative outcomes; non-linearity (spline on $S_B$); time-to-event analysis.
\end{itemize}

\section{Interpretability and reporting}
\begin{itemize}
    \item Coefficients $\boldsymbol{\gamma}$ as odds ratios; SHAP for the ML benchmark. \citeneeded{Lundberg and Lee 2017}
    \item Feature stability across resamples.
    \item Adherence to TRIPOD+AI \cite{collins_tripodai_2024} (Appendix~\ref{app:tripod}).
\end{itemize}
