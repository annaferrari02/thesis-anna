\section{Data and CFD pipeline}

\subsection{Cohort and outcome}
\begin{itemize}
    \item Patients treated with EVAR, with a valid preoperative CFD (Appendix~A) and benchmark features.
    \item Expected size: about 100--150 patients. {[}Open point: number of events.]
    \item Minimum follow-up (e.g.\ 12 months) is the only outcome-related exclusion: it avoids false negatives.
    \item Patients without complications are kept; imbalance is handled by the model intercept.
    \item One binary primary outcome $y_i \in \{0,1\}$, fixed in advance: any endoleak, or any complication.
    \item Subtypes (type I, type II) only as exploratory analyses.
\end{itemize}

\subsection{CFD pipeline}
\begin{enumerate}
    \item Segmentation of the lumen from preoperative CTA in 3D Slicer \cite{lasso_slicerheart_2022, trentin_automatic_2015}.
    \item Centerlines and clipping with VMTK. \citeneeded{VMTK}
    \item Mesh preprocessing (ParaView, pyvista) and volume meshing (SimVascular, TetGen). \citeneeded{SimVascular}
    \item Solution with svMultiPhysics. \citeneeded{svMultiPhysics}
    \item Quality control: periodicity of the last cycle, mass balance, plausible pressures (Appendix~A).
\end{enumerate}
\begin{itemize}
    \item Boundary conditions: inflow and RCR outflow \cite{du_outflow_2015, ramazanli_modeling_2025, les_quantification_2010}. {[}Open point: patient-specific or generic.]
    \item Rigid walls, Newtonian blood.
\end{itemize}

\subsection{Pre-specified protocol}
Outcome, regions, features, thresholds, number of components, folds and seeds are fixed before linking features to outcomes.

\section{Phase 1: tabular hemodynamic features}
Goal: test H1 with two nested logistic models that differ only by the block $\mathbf{Hd}$.

\subsection{Hemodynamic features}
\begin{itemize}
    \item Point indices TAWSS, OSI, RRT on the last cycle (Section~\ref{sec:wss-indices}).
    \item Anatomical regions $R_k$ from the centerline: proximal neck, sac, iliac landing zones.
\end{itemize}
\begin{equation}
    \bar{q}_{k} = \frac{1}{|R_k|}\int_{R_k} q(\mathbf{x})\,dA,
    \qquad
    f_{k}(\theta) = \frac{1}{|R_k|}\int_{R_k} \mathbb{1}\left[q(\mathbf{x}) > \theta\right] dA
\end{equation}
\begin{itemize}
    \item $\mathbf{Hd}_i \in \mathbb{R}^p$, with $p = 2$--$5$, chosen in advance from a mechanistic hypothesis, not from the outcome.
    \item Thresholds $\theta$ from the literature. \citeneeded{TAWSS and OSI thresholds}
    \item Features standardised before fitting.
\end{itemize}

\subsection{Benchmark score}
\begin{itemize}
    \item Benchmark: the supervisor's SVM on radiomics and morphology.
    \item Summarised in one number per patient: one parameter instead of many correlated features.
    \item Pre-validation: 5 stratified folds, $R = 10$--$20$ seeds; the whole benchmark pipeline (feature selection, tuning, SVM) is refitted inside each fold. \citeneeded{Tibshirani and Efron 2002}
    \item If the benchmark was developed on another cohort, its predictions are used directly.
\end{itemize}
\begin{equation}
    \bar{p}_i = \frac{1}{R}\sum_{r=1}^{R} \hat{p}_i^{(r)},
    \qquad
    S_{B,i} = \operatorname{logit}(\bar{p}_i) = \log\frac{\bar{p}_i}{1-\bar{p}_i}
\end{equation}

\subsection{Models and test}
\begin{align}
    M_0:\;& \operatorname{logit} P(y_i = 1) = \beta_0 + \beta_1 S_{B,i} \\
    M_1:\;& \operatorname{logit} P(y_i = 1) = \beta_0 + \beta_1 S_{B,i} + \boldsymbol{\gamma}^{\top}\mathbf{Hd}_i
\end{align}
\begin{itemize}
    \item $M_0$ is the logistic recalibration of the benchmark: same ranking, same AUC, better calibrated. \citeneeded{Cox 1958}
    \item $M_1$ is a model extension \cite{janssen_updating_2008}; $\beta_1$ is re-estimated together with $\boldsymbol{\gamma}$.
\end{itemize}
\begin{equation}
    H_0^{(1)}:\; \boldsymbol{\gamma} = \mathbf{0},
    \qquad
    \Lambda_1 = 2\left(\ell_1 - \ell_0\right) \;\overset{H_0}{\sim}\; \chi^2_{p}
\end{equation}
\begin{itemize}
    \item $\ell_j$: maximised log-likelihood of $M_j$, fitted on the same patients.
    \item Asymptotic $\chi^2_p$ by Wilks' theorem. \citeneeded{Wilks 1938}
    \item $\alpha = 0.05$; Holm--Bonferroni if H1 and H2 are both primary.
\end{itemize}

\section{Phase 2: geometry-aware representation}
Goal: test H2. Does the spatial distribution of the indices on the wall add information beyond $\mathbf{Hd}$?

\subsection{Point-based autoencoder}
\begin{itemize}
    \item Input: $N = 4096$ wall points per patient, sampled by area, aligned on the centerline.
    \item Per point: coordinates and $\mathbf{q} = (\log\mathrm{TAWSS}, \mathrm{OSI}, \log\mathrm{RRT})$, standardised.
    \item Encoder $E_\phi$: small PointNet++. \citeneeded{Qi et al.\ 2017, PointNet++}
    \item Decoder $D_\psi$: receives the latent code and a coordinate, returns the indices at that point. No outcome in the loss.
\end{itemize}
\begin{equation}
    \mathcal{P}_i = \left\{ \left(\mathbf{x}_{ij},\, \mathbf{q}_{ij}\right) \right\}_{j=1}^{N},
    \qquad
    \mathbf{z}_i = E_\phi(\mathcal{P}_i) \in \mathbb{R}^{32},
    \qquad
    \hat{\mathbf{q}}_{ij} = D_\psi(\mathbf{z}_i, \mathbf{x}_{ij})
\end{equation}
\begin{equation}
    \mathcal{L}(\phi,\psi) = \frac{1}{nN}\sum_{i=1}^{n}\sum_{j=1}^{N} \left\lVert \hat{\mathbf{q}}_{ij} - \mathbf{q}_{ij} \right\rVert^2_{\mathbf{W}}
\end{equation}
\begin{itemize}
    \item Checks: reconstruction $R^2$ on unseen patients; linear probe $\mathbf{Hd}_i \approx \mathbf{A}\mathbf{z}_i + \mathbf{b}$.
\end{itemize}

\subsection{Reduction}
\begin{equation}
    \mathbf{Z}_i = \mathbf{V}_k^{\top}\left(\bar{\mathbf{z}}_i - \boldsymbol{\mu}\right) \in \mathbb{R}^{k},
    \qquad k \in \{1, 2, 3\}
\end{equation}
\begin{itemize}
    \item $\bar{\mathbf{z}}_i$: mean of $\mathbf{z}_i$ over 10--20 point resamplings.
    \item $k$ fixed in advance, with $1 + p + k \le E/10$.
\end{itemize}

\subsection{Models and test}
\begin{align}
    M_2:\;& \operatorname{logit} P(y_i = 1) = \beta_0 + \beta_1 S_{B,i} + \boldsymbol{\delta}^{\top}\mathbf{Z}_i \\
    M_3:\;& \operatorname{logit} P(y_i = 1) = \beta_0 + \beta_1 S_{B,i} + \boldsymbol{\gamma}^{\top}\mathbf{Hd}_i + \boldsymbol{\delta}^{\top}\mathbf{Z}_i
\end{align}
\begin{equation}
    H_0^{(2)}:\; \boldsymbol{\delta} = \mathbf{0} \;\text{ in } M_3,
    \qquad
    \Lambda_2 = 2\left(\ell_3 - \ell_1\right) \;\overset{H_0}{\sim}\; \chi^2_{k}
\end{equation}
\begin{itemize}
    \item Main test: $M_3$ vs.\ $M_1$. Secondary: $M_2$ vs.\ $M_0$.
    \item Controls: $\mathbf{Z}_{\text{shape}}$ (geometric descriptors only, separates hemodynamics from shape); $\mathbf{Z}_{\text{perm}}$ (fields assigned to another patient's geometry, expected no gain).
\end{itemize}

\subsection{Interpretation}
\begin{equation}
    \hat{\mathbf{q}}(\mathbf{x}) = D_\psi\left(\boldsymbol{\mu} + a\,\sigma_c\,\mathbf{v}_c,\; \mathbf{x}\right),
    \qquad a \in \{-2, 0, 2\},\; \mathbf{x} \in \Gamma_{\mathrm{ref}}
\end{equation}
Each component is decoded on a reference geometry to show which spatial pattern it encodes.

\subsection{Optional: surface maps}
\begin{itemize}
    \item VMTK longitudinal--circumferential maps $m_i(s,\theta)$ and functional PCA give $\mathbf{Z}_{\text{map}}$. \citeneeded{VMTK branch mapping}
    \item $\mathbf{Z}$ vs.\ $\mathbf{Z}_{\text{map}}$: does the Euclidean neighbourhood of PointNet++ matter?
\end{itemize}

\section{Evaluation}

\subsection{Inference and performance chains}
\begin{itemize}
    \item Inference (LRT): models fitted on the full cohort; encoder and PCA fitted once without labels, so $\mathbf{Z}$ is consistent.
    \item Performance: the whole chain (encoder, PCA, logistic) refitted inside the same folds, giving out-of-fold predictions $\hat{\pi}^{(j)}_i$.
\end{itemize}

\subsection{Performance metrics}
\begin{equation}
    \Delta\mathrm{AUC} = \mathrm{AUC}\left(\hat{\pi}^{(1)}\right) - \mathrm{AUC}\left(\hat{\pi}^{(0)}\right),
    \qquad
    \Delta\mathrm{BS} = \frac{1}{n}\sum_{i=1}^{n}\left[\left(\hat{\pi}_i^{(1)} - y_i\right)^2 - \left(\hat{\pi}_i^{(0)} - y_i\right)^2\right]
\end{equation}
\begin{equation}
    \mathrm{NB}(p_t) = \frac{\mathrm{TP}(p_t)}{n} - \frac{\mathrm{FP}(p_t)}{n}\cdot\frac{p_t}{1 - p_t}
    \quad \citeneeded{Vickers and Elkin 2006}
\end{equation}
\begin{itemize}
    \item Calibration intercept and slope out of fold \cite{alba_discrimination_2017}.
    \item 95\% CIs by paired bootstrap ($B = 2000$). Effect sizes, not tests.
\end{itemize}

\subsection{Sample size}
\begin{equation}
    1 + p + k \;\le\; \frac{E}{10}
    \quad \citeneeded{Peduzzi et al.\ 1996}
\end{equation}
\begin{itemize}
    \item $E$: number of events. With $E = 40$ and Phase~1 only, at most $p = 3$.
    \item The 10-events rule is a heuristic. \citeneeded{Riley et al.\ 2020}
\end{itemize}

\subsection{Sensitivity analyses}
\begin{itemize}
    \item Spline on $S_B$: linearity in the logit.
    \item Firth regression: small-sample bias. \citeneeded{Firth 1993}
    \item Parametric bootstrap of the LRT: $\chi^2$ approximation. \citeneeded{Höfling and Tibshirani 2008}
    \item Wald test, number of seeds, follow-up threshold.
    \item Exploratory: $S_B \times \mathbf{Hd}$ interaction, other outcomes.
\end{itemize}
