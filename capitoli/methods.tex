\section{Reference model and score}
\begin{itemize}
    \item Reproduce the benchmark pipeline (Appendix~\ref{app:benchmark}).
    \item Out-of-fold score: 5 stratified folds, $R$ seeds, whole pipeline refitted in each fold. (we use out of fold pz as a sort of pre validation set per model extension)
    \item Why out-of-fold: in-sample scores are optimistic and would bias the test against $\mathbf{Hd}$. So, we train and test the benchmark and obtain an OOF score per patient which summarizes the prediction of the model on unseen patient (average of the p assigned to each patient in each repetition): 
\end{itemize}
\begin{equation}
    \bar{p}_i = \frac{1}{R}\sum_{r=1}^{R} \hat{p}_i^{(r)},
    \qquad
    S_{B,i} = \operatorname{logit}(\bar{p}_i)
\end{equation}

\section{Primary analysis}

\subsection{Models}
\begin{align}
    M_0:\;& \operatorname{logit} P(D_i = 1) = \beta_0 + \beta_1 S_{B,i} \\
    M_1:\;& \operatorname{logit} P(D_i = 1) = \beta_0 + \beta_1 S_{B,i} + \boldsymbol{\gamma}^{\top}\mathbf{Hd}_i
\end{align}
$M_0$ is the recalibrated benchmark (recalibration is required for comparing two models discrimination, otherwise we mix the effects) \citeneeded{Cox 1958}; $M_1$ extends it with $\mathbf{Hd}$ \cite{janssen_updating_2008}.

\begin{enumerate}
    \item \textbf{Calibration} of both models: intercept and slope \cite{alba_discrimination_2017}. 
    \item \textbf{Likelihood ratio test} on the $\mathbf{Hd}$ block \cite{pepe_testing_2013}:
    \begin{equation}
        H_0:\; \boldsymbol{\gamma} = \mathbf{0},
        \qquad
        \Lambda = 2\left(\ell_1 - \ell_0\right) \;\overset{H_0}{\sim}\; \chi^2_{p}
    \end{equation}
    \item \textbf{Performance gain} with bootstrap CI: $\Delta$AUC, $\Delta$Brier, $\Delta$net benefit. Pepe 2013 proved that if we reject H0 we confirm also a statistically significant increase in $\Delta$AUC, but we still need to QUANTIFY the improvement. this is what performance gain is for. 
    
    \item \textbf{Correction for optimism???}
\end{enumerate}

\section{Secondary analysis (Q2)}

\subsection{Proposed Model structure (TBD)}
\begin{itemize}
    \item Why: regional features average each region and lose \emph{where} the values are on the wall.
    \item Input: each patient as a set of wall points with coordinates and indices, $\mathcal{P}_i = \{(\mathbf{x}_{ij}, \mathbf{q}_{ij})\}_{j=1}^{N}$.
    \item Encoder: small PointNet++ \citeneeded{Qi et al.\ 2017, PointNet++}, as in Peng et al.\ \citeneeded{Peng et al.\ 2026}. Same layers on local neighbourhoods, then pooling: output independent of point order, dependent on the spatial arrangement of the values.
    \item Autoencoder, trained without the outcome: the decoder gets $\mathbf{z}_i$ and a coordinate and reconstructs the indices there.
    \item The decoder already has the geometry $\rightarrow$ $\mathbf{z}_i$ encodes how the field is distributed on the wall, not the shape.
    \item Reduction: 32 dimensions too many for few events $\rightarrow$ average over point resamplings, PCA to $k = 1$--$3$ components $\mathbf{Z}_i$.
    \item $\mathbf{Z}_i$ enters the same logistic model as Q1, as one more block: spatial information learned without labels, prediction still a small logistic model, comparable with $M_1$.

\end{itemize}
